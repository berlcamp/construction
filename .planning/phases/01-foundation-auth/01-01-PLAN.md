2. Create `supabase/migrations/001_create_construction_schema_and_profiles.sql`:
   ```sql
   -- Migration 001: Create construction schema and profiles table
   -- Per D-PS-09: Only tables needed for auth/onboarding in Phase 1

   CREATE SCHEMA IF NOT EXISTS construction;

   -- Grant usage to authenticated and anon roles so PostgREST can access
   GRANT USAGE ON SCHEMA construction TO anon, authenticated, service_role;

   -- Profiles table: auto-populated by trigger on auth.users
   CREATE TABLE construction.profiles (
     id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
     email TEXT NOT NULL,
     full_name TEXT,
     avatar_url TEXT,
     phone TEXT,
     created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
     updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
   );

   -- Grant table permissions
   GRANT ALL ON ALL TABLES IN SCHEMA construction TO anon, authenticated, service_role;
   GRANT ALL ON ALL SEQUENCES IN SCHEMA construction TO anon, authenticated, service_role;
   GRANT ALL ON ALL ROUTINES IN SCHEMA construction TO anon, authenticated, service_role;

   -- Set default privileges for future tables in the schema
   ALTER DEFAULT PRIVILEGES IN SCHEMA construction GRANT ALL ON TABLES TO anon, authenticated, service_role;
   ALTER DEFAULT PRIVILEGES IN SCHEMA construction GRANT ALL ON SEQUENCES TO anon, authenticated, service_role;
   ALTER DEFAULT PRIVILEGES IN SCHEMA construction GRANT ALL ON ROUTINES TO anon, authenticated, service_role;

   -- updated_at trigger function (reusable for all tables)
   CREATE OR REPLACE FUNCTION construction.set_updated_at()
   RETURNS TRIGGER AS $$
   BEGIN
     NEW.updated_at = now();
     RETURN NEW;
   END;
   $$ LANGUAGE plpgsql;

   -- Apply updated_at trigger to profiles
   CREATE TRIGGER set_profiles_updated_at
     BEFORE UPDATE ON construction.profiles
     FOR EACH ROW
     EXECUTE FUNCTION construction.set_updated_at();
   ```

3. Create `supabase/migrations/002_handle_new_user_trigger.sql`:
   ```sql
   -- Migration 002: Auto-create profile when a new user signs up
   -- Per FOUND-05: profile INSERT trigger runs as SECURITY DEFINER
   -- This function runs on auth.users INSERT (e.g., after Google OAuth sign-up)

   CREATE OR REPLACE FUNCTION public.handle_new_user()
   RETURNS TRIGGER
   SECURITY DEFINER
   SET search_path = ''
   AS $$
   BEGIN
     INSERT INTO construction.profiles (id, email, full_name, avatar_url)
     VALUES (
       NEW.id,
       NEW.email,
       COALESCE(NEW.raw_user_meta_data ->> 'full_name', NEW.raw_user_meta_data ->> 'name', ''),
       COALESCE(NEW.raw_user_meta_data ->> 'avatar_url', NEW.raw_user_meta_data ->> 'picture', '')
     );
     RETURN NEW;
   END;
   $$ LANGUAGE plpgsql;

   -- Trigger on auth.users table
   CREATE TRIGGER on_auth_user_created
     AFTER INSERT ON auth.users
     FOR EACH ROW
     EXECUTE FUNCTION public.handle_new_user();
   ```
   NOTE: The function is in `public` schema (not `construction`) because Supabase trigger functions on `auth.users` must be in `public`. It uses `SECURITY DEFINER` so it can write to `construction.profiles` regardless of the invoking role. `SET search_path = ''` prevents search_path attacks.

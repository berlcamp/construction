    ```typescript
    'use server'

    import { createClient } from '@/lib/supabase/server'
    import { createCompanySchema } from '@/lib/validations/company.schema'
    import { redirect } from 'next/navigation'

    export async function createCompany(formData: FormData) {
      const supabase = await createClient()
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) throw new Error('Not authenticated')

      // Check if user already has a company
      const { data: existingMembership } = await supabase
        .from('company_members')
        .select('id')
        .eq('user_id', user.id)
        .maybeSingle()
      if (existingMembership) redirect('/dashboard')

      const input = createCompanySchema.parse({
        name: formData.get('name'),
        address: formData.get('address') || '',
        phone: formData.get('phone') || '',
      })

      // Generate slug from company name
      const slug = input.name
        .toLowerCase()
        .replace(/[^a-z0-9]+/g, '-')
        .replace(/^-|-$/g, '')
        + '-' + Date.now().toString(36)

      // 1. Create company
      const { data: company, error: companyError } = await supabase
        .from('companies')
        .insert({
          name: input.name,
          slug,
          owner_id: user.id,
          address: input.address || null,
          contact_phone: input.phone || null,
          contact_email: user.email,
        })
        .select('id')
        .single()

      if (companyError) throw new Error(`Failed to create company: ${companyError.message}`)

      // 2. Create owner membership
      const { error: memberError } = await supabase
        .from('company_members')
        .insert({
          company_id: company.id,
          user_id: user.id,
          role: 'owner',
        })

      if (memberError) throw new Error(`Failed to create membership: ${memberError.message}`)

      // 3. Create trial subscription (14 days, professional tier limits per BILL-03)
      const now = new Date()
      const trialEnd = new Date(now)
      trialEnd.setDate(trialEnd.getDate() + 14)

      const { error: subError } = await supabase
        .from('subscriptions')
        .insert({
          company_id: company.id,
          plan: 'professional',
          amount: 0,
          period_start: now.toISOString().split('T')[0],
          period_end: trialEnd.toISOString().split('T')[0],
          status: 'paid',
        })

      if (subError) {
        console.error('Subscription creation failed:', subError)
        // Non-blocking — company + membership already created
      }

      redirect('/dashboard')
    }
    ```

    NOTE: The company creation uses the Supabase server client (cookie-based, respects RLS).     **2. src/app/(app)/settings/members/actions.ts** — Server Action to send invite (AUTH-07):

    ```typescript
    'use server'
    import { createClient } from '@/lib/supabase/server'
    import { getCompanyContext } from '@/lib/auth/getCompanyContext'
    import { inviteUserSchema } from '@/lib/validations/invite.schema'
    import { revalidatePath } from 'next/cache'

    export async function inviteUser(formData: FormData) {
      const supabase = await createClient()
      const ctx = await getCompanyContext(supabase)
      if (!ctx) throw new Error('Not authenticated')
      if (!['owner', 'admin'].includes(ctx.role)) {
        throw new Error('Only owners and admins can invite users')
      }

      const input = inviteUserSchema.parse({
        email: formData.get('email'),
        role: formData.get('role'),
      })

      const { data: invitation, error } = await supabase
        .from('company_invitations')
        .insert({
          company_id: ctx.companyId,
          invited_email: input.email,
          role: input.role,
          invited_by: ctx.userId,
        })
        .select('id, token')
        .single()

      if (error) {
        if (error.code === '23505') {
          throw new Error('A pending invitation for this email already exists')
        }
        throw new Error(`Failed to send invitation: ${error.message}`)
      }

      // For MVP: log the invite link to console. In production, send via Resend.
      const inviteUrl = `${process.env.NEXT_PUBLIC_APP_URL}/invite/${invitation.token}`
      console.log(`Invite URL for ${input.email}: ${inviteUrl}`)

      revalidatePath('/settings/members')
      return { success: true, inviteUrl }
    }
    ```

    **6. src/app/invite/[token]/actions.ts** — Server Action to accept invitation:
    ```typescript
    'use server'
    import { createClient } from '@/lib/supabase/server'
    import { redirect } from 'next/navigation'

    export async function acceptInvitation(token: string) {
      const supabase = await createClient()
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) throw new Error('Not authenticated')

      // Find the invitation
      const { data: invitation, error: findError } = await supabase
        .from('company_invitations')
        .select('id, company_id, role, invited_email, expires_at, status')
        .eq('token', token)
        .single()

      if (!invitation || findError) redirect('/invite/error')

      // Validate: pending + not expired (AUTH-09)
      if (invitation.status !== 'pending') redirect('/invite/error')
      if (new Date(invitation.expires_at) < new Date()) {
        // Update status to expired
        await supabase
          .from('company_invitations')
          .update({ status: 'expired' })
          .eq('id', invitation.id)
        redirect('/invite/error')
      }

      // Check user doesn't already belong to a company
      const { data: existing } = await supabase
        .from('company_members')
        .select('id')
        .eq('user_id', user.id)
        .maybeSingle()

      if (existing) {
        throw new Error('You already belong to a company')
      }

      // Create company_members record (AUTH-08)
      const { error: memberError } = await supabase
        .from('company_members')
        .insert({
          company_id: invitation.company_id,
          user_id: user.id,
          role: invitation.role,
        })

      if (memberError) throw new Error(`Failed to accept invitation: ${memberError.message}`)

      // Update invitation status
      await supabase
        .from('company_invitations')
        .update({ status: 'accepted', accepted_at: new Date().toISOString() })
        .eq('id', invitation.id)

      redirect('/dashboard')
    }
    ```

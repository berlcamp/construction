    ```typescript
    'use server'

    import { createClient } from '@/lib/supabase/server'
    import { createCompanySchema } from '@/lib/validations/company.schema'
    import { redirect } from 'next/navigation'

    export async function createCompany(formData: FormData) {
      const supabase = await createClient()
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) throw new Error('Not authenticated')

      // Check if user already has a company
      const { data: existingMembership } = await supabase
        .from('company_members')
        .select('id')
        .eq('user_id', user.id)
        .maybeSingle()
      if (existingMembership) redirect('/dashboard')

      const input = createCompanySchema.parse({
        name: formData.get('name'),
        address: formData.get('address') || '',
        phone: formData.get('phone') || '',
      })

      // Generate slug from company name
      const slug = input.name
        .toLowerCase()
        .replace(/[^a-z0-9]+/g, '-')
        .replace(/^-|-$/g, '')
        + '-' + Date.now().toString(36)

      // 1. Create company
      const { data: company, error: companyError } = await supabase
        .from('companies')
        .insert({
          name: input.name,
          slug,
          owner_id: user.id,
          address: input.address || null,
          contact_phone: input.phone || null,
          contact_email: user.email,
        })
        .select('id')
        .single()

      if (companyError) throw new Error(`Failed to create company: ${companyError.message}`)

      // 2. Create owner membership
      const { error: memberError } = await supabase
        .from('company_members')
        .insert({
          company_id: company.id,
          user_id: user.id,
          role: 'owner',
        })

      if (memberError) throw new Error(`Failed to create membership: ${memberError.message}`)

      // 3. Create trial subscription (14 days, professional tier limits per BILL-03)
      const now = new Date()
      const trialEnd = new Date(now)
      trialEnd.setDate(trialEnd.getDate() + 14)

      const { error: subError } = await supabase
        .from('subscriptions')
        .insert({
          company_id: company.id,
          plan: 'professional',
          amount: 0,
          period_start: now.toISOString().split('T')[0],
          period_end: trialEnd.toISOString().split('T')[0],
          status: 'paid',
        })

      if (subError) {
        console.error('Subscription creation failed:', subError)
        // Non-blocking — company + membership already created
      }

      redirect('/dashboard')
    }
    ```

    CRITICAL: Per D-10, there is NO pre-accept display page. The invite link redirects DIRECTLY to Google OAuth. No page.tsx for /invite/[token] — only a route.ts (Route Handler).

    **The direct-to-OAuth invite flow works as follows:**
    1. User clicks invite link: `/invite/{token}`
    2. Route Handler validates token (exists, pending, not expired)
    3. If invalid/expired -> redirect to `/invite/error?reason=invalid` or `/invite/error?reason=expired`
    4. If valid -> redirect to Supabase Google OAuth URL with `state=invite:{token}` encoded in the options
    5. After Google OAuth, callback route reads the state param
    6. If state starts with `invite:`, extract token and call `acceptInvitation(token)`
    7. `acceptInvitation` validates token again, creates company_members, marks invitation accepted
    8. Redirect to `/dashboard`

    ```typescript
    'use server'
    import { createClient } from '@/lib/supabase/server'
    import { getCompanyContext } from '@/lib/auth/getCompanyContext'
    import { inviteUserSchema } from '@/lib/validations/invite.schema'
    import { revalidatePath } from 'next/cache'

    export async function inviteUser(formData: FormData) {
      const supabase = await createClient()
      const ctx = await getCompanyContext(supabase)
      if (!ctx) throw new Error('Not authenticated')
      if (!['owner', 'admin'].includes(ctx.role)) {
        throw new Error('Only owners and admins can invite users')
      }

      const input = inviteUserSchema.parse({
        email: formData.get('email'),
        role: formData.get('role'),
      })

      const { data: invitation, error } = await supabase
        .from('company_invitations')
        .insert({
          company_id: ctx.companyId,
          invited_email: input.email,
          role: input.role,
          invited_by: ctx.userId,
        })
        .select('id, token')
        .single()

      if (error) {
        if (error.code === '23505') {
          throw new Error('A pending invitation for this email already exists')
        }
        throw new Error(`Failed to send invitation: ${error.message}`)
      }

      // For MVP: log the invite link to console. In production, send via Resend.
      const inviteUrl = `${process.env.NEXT_PUBLIC_APP_URL}/invite/${invitation.token}`
      console.log(`Invite URL for ${input.email}: ${inviteUrl}`)

      revalidatePath('/settings/members')
      return { success: true, inviteUrl }
    }
    ```

    **5. src/app/invite/[token]/route.ts** — Route Handler for invite link (per D-10: NO page, direct redirect):
    This is a Next.js Route Handler (NOT a page component). There is NO page.tsx in this directory.
    ```typescript
    import { NextResponse } from 'next/server'
    import { createClient } from '@supabase/supabase-js'

    export async function GET(
      request: Request,
      { params }: { params: Promise<{ token: string }> }
    ) {
      const { token } = await params
      const origin = new URL(request.url).origin

      // Use service-role client to validate token without auth requirement
      // (user is not logged in yet at this point)
      const supabase = createClient(
        process.env.NEXT_PUBLIC_SUPABASE_URL!,
        process.env.SUPABASE_SERVICE_ROLE_KEY!,
        { db: { schema: 'construction' } }
      )

      // Validate the invitation token
      const { data: invitation, error } = await supabase
        .from('company_invitations')
        .select('id, status, expires_at')
        .eq('token', token)
        .single()

      if (error || !invitation) {
        return NextResponse.redirect(`${origin}/invite/error?reason=invalid`)
      }

      if (invitation.status !== 'pending') {
        return NextResponse.redirect(`${origin}/invite/error?reason=used`)
      }

      if (new Date(invitation.expires_at) < new Date()) {
        // Mark as expired
        await supabase
          .from('company_invitations')
          .update({ status: 'expired' })
          .eq('id', invitation.id)
        return NextResponse.redirect(`${origin}/invite/error?reason=expired`)
      }

      // Token is valid — redirect directly to Google OAuth with invite token in state
      // The state param will be passed through OAuth and available in the callback
      const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL!
      const redirectTo = `${origin}/auth/callback`

      // Build the Supabase OAuth URL directly
      // Supabase OAuth endpoint: {supabase_url}/auth/v1/authorize
      const oauthUrl = new URL(`${supabaseUrl}/auth/v1/authorize`)
      oauthUrl.searchParams.set('provider', 'google')
      oauthUrl.searchParams.set('redirect_to', redirectTo)
      // Encode the invite token in a custom query param that Supabase passes through
      // We use the `state` approach: append to redirect_to as a query param
      // so the callback receives it
      const callbackWithState = `${redirectTo}?invite_token=${encodeURIComponent(token)}`
      oauthUrl.searchParams.set('redirect_to', callbackWithState)

      return NextResponse.redirect(oauthUrl.toString())
    }
    ```

    IMPORTANT IMPLEMENTATION NOTE: The exact mechanism for passing the invite token through OAuth depends on the Supabase auth API. The above uses `redirect_to` with a query param appended. An alternative approach is to use Supabase's `signInWithOAuth` options from the server side. During execution, verify which approach works:

    Option A (preferred — simpler): Append `?invite_token={token}` to the `redirect_to` URL. Supabase will redirect back to `/auth/callback?code=xxx&invite_token=yyy`.

    Option B (fallback): If Supabase strips extra query params from redirect_to, store the invite token in a short-lived httpOnly cookie before redirecting to OAuth, then read the cookie in the callback.

    The executor should test Option A first and fall back to Option B if needed.

    ```typescript
    import { createClient } from '@/lib/supabase/server'

    export async function GET(request: Request) {
      const { searchParams, origin } = new URL(request.url)
      const code = searchParams.get('code')
      const inviteToken = searchParams.get('invite_token')
      const next = searchParams.get('next') ?? '/dashboard'

      if (code) {
        const supabase = await createClient()
        const { error } = await supabase.auth.exchangeCodeForSession(code)
        if (!error) {
          const { data: { user } } = await supabase.auth.getUser()
          if (user) {

            if (inviteToken) {
              const result = await acceptInvitation(supabase, inviteToken, user.id)
              if (result.error) {
                return NextResponse.redirect(`${origin}/invite/error?reason=${result.error}`)
              }
              // Successfully accepted — redirect to dashboard
              return NextResponse.redirect(`${origin}/dashboard`)
            }

            // Normal login flow — check for existing membership
            const { data: membership } = await supabase
              .from('company_members')
              .select('id')
              .eq('user_id', user.id)
              .eq('is_active', true)
              .maybeSingle()

            if (!membership) {
              // No company — redirect to onboarding (per AUTH-03)
              return NextResponse.redirect(`${origin}/onboarding`)
            }
          }
          return NextResponse.redirect(`${origin}${next}`)
        }
      }

      // Auth error — redirect to login with error
      return NextResponse.redirect(`${origin}/login?error=auth_callback_failed`)
    }
    ```

    **7. src/lib/auth/acceptInvitation.ts** — Shared invitation acceptance logic:
    ```typescript
    import { SupabaseClient } from '@supabase/supabase-js'

    interface AcceptResult {
      success: boolean
      error?: 'invalid' | 'expired' | 'used' | 'already_member' | 'failed'
    }

    export async function acceptInvitation(
      supabase: SupabaseClient,
      token: string,
      userId: string
    ): Promise<AcceptResult> {
      // Find the invitation
      const { data: invitation, error: findError } = await supabase
        .from('company_invitations')
        .select('id, company_id, role, invited_email, expires_at, status')
        .eq('token', token)
        .single()

      if (!invitation || findError) {
        return { success: false, error: 'invalid' }
      }

      // Validate: pending + not expired (AUTH-09)
      if (invitation.status !== 'pending') {
        return { success: false, error: 'used' }
      }

      if (new Date(invitation.expires_at) < new Date()) {
        await supabase
          .from('company_invitations')
          .update({ status: 'expired' })
          .eq('id', invitation.id)
        return { success: false, error: 'expired' }
      }

      // Check user doesn't already belong to a company
      const { data: existing } = await supabase
        .from('company_members')
        .select('id')
        .eq('user_id', userId)
        .maybeSingle()

      if (existing) {
        return { success: false, error: 'already_member' }
      }

      // Create company_members record (AUTH-08)
      const { error: memberError } = await supabase
        .from('company_members')
        .insert({
          company_id: invitation.company_id,
          user_id: userId,
          role: invitation.role,
        })

      if (memberError) {
        return { success: false, error: 'failed' }
      }

      // Update invitation status
      await supabase
        .from('company_invitations')
        .update({ status: 'accepted', accepted_at: new Date().toISOString() })
        .eq('id', invitation.id)

      return { success: true }
    }
    ```

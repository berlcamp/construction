-- Migration 004: Create company_members junction table
-- Per D-13: company_members is the tenancy source of truth
-- Per AUTH-10: Partial unique index enforces one owner per company
-- Per Pitfall 2: Composite index on (user_id, company_id) for RLS performance

CREATE TABLE construction.company_members (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES construction.profiles(id) ON DELETE CASCADE,
  company_id UUID NOT NULL REFERENCES construction.companies(id) ON DELETE CASCADE,
  role TEXT NOT NULL DEFAULT 'staff'
    CHECK (role IN ('owner', 'admin', 'project_manager', 'accountant', 'procurement_officer', 'staff')),
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Composite index for RLS policy performance (Pitfall 2)
CREATE INDEX idx_company_members_user_company
  ON construction.company_members(user_id, company_id);

-- Unique constraint: a user can only be a member of a company once
CREATE UNIQUE INDEX idx_company_members_user_company_unique
  ON construction.company_members(user_id, company_id);

-- Partial unique index: exactly one owner per company (AUTH-10)
CREATE UNIQUE INDEX idx_one_owner_per_company
  ON construction.company_members(company_id)
  WHERE role = 'owner';

-- Apply updated_at trigger
CREATE TRIGGER set_company_members_updated_at
  BEFORE UPDATE ON construction.company_members
  FOR EACH ROW
  EXECUTE FUNCTION construction.set_updated_at();
```

-- Migration 005: Create company_invitations table
-- Per AUTH-07: Owner/Admin can invite users by email with role assignment
-- Per AUTH-09: Invitations expire after 7 days; duplicate pending invitations rejected

CREATE TABLE construction.company_invitations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id UUID NOT NULL REFERENCES construction.companies(id) ON DELETE CASCADE,
  invited_by UUID NOT NULL REFERENCES construction.profiles(id) ON DELETE CASCADE,
  invited_email TEXT NOT NULL,
  role TEXT NOT NULL DEFAULT 'staff'
    CHECK (role IN ('admin', 'project_manager', 'accountant', 'procurement_officer', 'staff')),
  token UUID NOT NULL DEFAULT gen_random_uuid(),
  expires_at TIMESTAMPTZ NOT NULL DEFAULT (now() + INTERVAL '7 days'),
  status TEXT NOT NULL DEFAULT 'pending'
    CHECK (status IN ('pending', 'accepted', 'expired', 'cancelled')),
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Unique token for lookup
CREATE UNIQUE INDEX idx_invitation_token ON construction.company_invitations(token);

-- Prevent duplicate pending invitations to the same email for the same company (AUTH-09)
CREATE UNIQUE INDEX idx_one_pending_invite_per_email_company
  ON construction.company_invitations(company_id, invited_email)
  WHERE status = 'pending';
```

NOTE: The `role` CHECK on invitations excludes 'owner' — owners cannot be invited, they create the company. Expires at defaults to 7 days from creation per AUTH-09.
